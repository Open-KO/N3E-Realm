cmake_minimum_required(VERSION 3.25)
project(n3e VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(N3E_TOOLS "Compile local tools UI" OFF)
# Linux takes SDL3 from pkg-config and protobuf from CMake's module; Windows takes
# both, OpenSSL and the Khronos GL headers from vcpkg (vcpkg.json, spec 04 "Build"),
# linked statically with the static C runtime (triplet x64-windows-static), so the
# client is one .exe with no Visual C++ runtime to install.
add_library(n3e-sdl3 INTERFACE)
if(WIN32)
  find_package(SDL3 CONFIG REQUIRED)
  target_link_libraries(n3e-sdl3 INTERFACE SDL3::SDL3)
  find_package(Protobuf CONFIG REQUIRED)
  find_path(N3E_GLCOREARB GL/glcorearb.h REQUIRED)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL3 REQUIRED IMPORTED_TARGET sdl3>=3.2)
  target_link_libraries(n3e-sdl3 INTERFACE PkgConfig::SDL3)
  find_package(Protobuf REQUIRED)
endif()
find_package(OpenGL REQUIRED)
find_package(OpenSSL REQUIRED)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set(N3E_WARNINGS /W3 /utf-8 /bigobj)
  # Winsock and <windows.h> without min/max macros; M_PI from <cmath>; the CRT's
  # getenv/sscanf without deprecation notes.
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _USE_MATH_DEFINES _CRT_SECURE_NO_WARNINGS)
  # The client's own sources (not protobuf's): Windows headers first, minus the
  # macros that collide with its names (client/platform/windows.h).
  set(N3E_WINDOWS_FIRST "/FI${CMAKE_CURRENT_SOURCE_DIR}/client/platform/windows.h")
  list(APPEND N3E_WARNINGS ${N3E_WINDOWS_FIRST})
else()
  set(N3E_WARNINGS -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  add_library(n3e-wire STATIC)
  protobuf_generate(TARGET n3e-wire LANGUAGE cpp PROTOS protocol/wire.proto IMPORT_DIRS protocol
                    PROTOC_OUT_DIR ${CMAKE_CURRENT_BINARY_DIR})
else()
  protobuf_generate_cpp(WIRE_SRC WIRE_HDR protocol/wire.proto)
  add_library(n3e-wire STATIC ${WIRE_SRC})
endif()
target_include_directories(n3e-wire PUBLIC ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(n3e-wire PUBLIC protobuf::libprotobuf)
find_package(Threads REQUIRED)
add_library(n3e-assets STATIC client/assets/assets.cpp client/assets/collision.cpp client/assets/world_stream.cpp)
target_include_directories(n3e-assets PUBLIC client)
target_compile_options(n3e-assets PRIVATE ${N3E_WINDOWS_FIRST})
target_link_libraries(n3e-assets PUBLIC OpenSSL::Crypto Threads::Threads)
# Two clients from one source (PQ-006). n3e-client is the player build: its replay,
# smoke and capture hooks are compiled out (N3E_HARNESS=0, client/harness/harness.h)
# and tests/player_build.py checks none of their strings reach the binary.
# n3e-test-client (N3E_HARNESS=1) adds them; every test and developer session uses it.
add_library(n3e-client-network OBJECT client/network/handshake.cpp client/network/world.cpp client/network/movement.cpp client/network/discord_login.cpp client/network/websocket.cpp)
target_link_libraries(n3e-client-network PRIVATE n3e-assets n3e-wire n3e-sdl3 OpenSSL::SSL)
target_compile_options(n3e-client-network PRIVATE ${N3E_WARNINGS})
if(N3E_TOOLS)
  if(NOT EXISTS "${IMGUI_DIR}/imgui.cpp")
    message(FATAL_ERROR "Run tools/bootstrap.sh and source tools/env.sh for pinned ImGui")
  endif()
  execute_process(COMMAND git -C "${IMGUI_DIR}" rev-parse HEAD OUTPUT_VARIABLE IMGUI_REV OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT IMGUI_REV STREQUAL "f5befd2d29e66809cd1110a152e375a7f1981f06")
    message(FATAL_ERROR "ImGui revision mismatch")
  endif()
  add_library(n3e-imgui OBJECT ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp ${IMGUI_DIR}/backends/imgui_impl_sdl3.cpp ${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp)
  target_include_directories(n3e-imgui PUBLIC ${IMGUI_DIR} ${IMGUI_DIR}/backends)
  target_link_libraries(n3e-imgui PUBLIC n3e-sdl3 OpenGL::GL)
endif()
foreach(client n3e-client n3e-test-client)
  add_executable(${client} client/main.cpp client/self_test.cpp client/render/viewer.cpp $<TARGET_OBJECTS:n3e-client-network>)
  target_link_libraries(${client} PRIVATE n3e-assets n3e-wire n3e-sdl3 OpenGL::GL OpenSSL::SSL)
  if(WIN32)
    # Winsock, the ROOT certificate store, ShellExecute; the manifest makes the
    # process code page UTF-8, so argv, paths and getenv carry any user name.
    target_sources(${client} PRIVATE client/platform/n3e-client.manifest)
    target_include_directories(${client} PRIVATE ${N3E_GLCOREARB})
    target_link_libraries(${client} PRIVATE ws2_32 crypt32 shell32)
  else()
    # tools/package_client.py puts the libraries a stock distribution lacks beside it.
    set_target_properties(${client} PROPERTIES BUILD_RPATH "\$ORIGIN")
  endif()
  if(client STREQUAL "n3e-test-client")
    set(harness 1)
  else()
    set(harness 0)
  endif()
  target_compile_definitions(${client} PRIVATE TOOLS=$<BOOL:${N3E_TOOLS}> N3E_HARNESS=${harness})
  target_compile_options(${client} PRIVATE ${N3E_WARNINGS})
  if(N3E_TOOLS)
    target_link_libraries(${client} PRIVATE n3e-imgui)
  endif()
endforeach()
if(WIN32)
  return() # the checks and the test peer are Linux tools (make test)
endif()
